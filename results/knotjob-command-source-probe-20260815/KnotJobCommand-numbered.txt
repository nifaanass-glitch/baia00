   175	    private void loadTheLinks() {
   176	        for (int i = 0; i < fileNames.size(); i++) {
   177	            File[] files = new File[1];
   178	            files[0] = new File(fileNames.get(i));
   179	            countDown = new CountDownLatch(1);
   180	            LinkLoader loader = new LinkLoader(files, countDown);
   181	            loader.start();
   182	            try {
   183	                countDown.await();
   184	            }
   185	            catch (InterruptedException ex) {
   186	                Logger.getLogger(KnotJobCommand.class.getName()).log(Level.SEVERE, null, ex);
   187	            }
   188	            theLinks.add(loader.getLinks());
   189	        } 
   190	    }

   231	    private void checkSInvariant(String command) {
   232	        if ("-sqo".equals(command) || "-sq1o".equals(command)) {
   233	            sqOneOdd = true;
   234	            return;
   235	        }
   236	        if ("-sqe".equals(command) || "-sq1e".equals(command)) {
   237	            sqOneEven = true;
   238	            return;
   239	        }
   240	        if ("-sqs".equals(command) || "-sq1s".equals(command)) {
   241	            sqOneSum = true;
   242	            return;
   243	        }
   244	        if ("-scr".equals(command)) {
   245	            comr = true;
   246	            return;
   247	        }
   248	        if ("-sq2e".equals(command)) {
   249	            sqTwoEven = true;
   250	            return;
   251	        }
   252		if ("-sq2o0".equals(command)) {
   253	            sqTwoOdd = true;
   254	            return;
   255	        }
   256	        if ("-sq2o1".equals(command)) {
   257	            sqTwoOde = true;
   258	            return;
   259	        }
   260	        if (command.length()>=5 && "-sbls".equals(command.substring(0, 5))) {
   261	            blsOdd = true;
   262	            int pwr = 3;
   263	            try {
   264	                pwr = Integer.parseInt(command.substring(5));
   265	            }
   266	            catch(NumberFormatException ex) {
   267	                // pwr will remain 3
   268	            }
   269	            if (pwr < 3) pwr = 3;
   270	            if (pwr > 15) pwr = 15;
   271	            blsmod = 2; 
   272	            for (int i = 1; i < pwr; i++) blsmod = blsmod*2;
   273	            return;
   274	        }
   275		if ("-sgr".equals(command)) {
   276	            sGraded = true;
   277	            return;
   278	        }
   279	        if ("-sgn".equals(command)) {
   280	            signature = true;
   281	            return;
   282	        }
   283	        if (command.startsWith("-sl")) {
   284	            int characteristic = getSNumber(command.substring(1));
   285	            if (characteristic >= 0 && !slInvs.contains(characteristic)) slInvs.add(characteristic);
   286	            return;
   287	        }
   288	        int characteristic = getSNumber(command);
   289	        if (characteristic >= 0 && !sInvs.contains(characteristic)) sInvs.add(characteristic);
   290	    }

   334	    private int getSNumber(String command) {
   335	        int ch;
   336	        try {
   337	            ch = Integer.parseInt(command.substring(2));
   338	        }
   339	        catch (NumberFormatException ne) {
   340	            ch = -1;
   341	        }
   342	        if (ch <= 0) return ch;
   343	        if (acceptedPrime(ch)) return ch;
   344	        return -1;
   345	    }

   347	    private boolean acceptedPrime(int ch) {
   348	        boolean accepted = false;
   349	        int i = 0;
   350	        while (!accepted && i < options.getPrimes().size()) {
   351	            if (ch == options.getPrimes().get(i)) accepted = true;
   352	            else i++;
   353	        }
   354	        return accepted;
   355	    }
   356	
   357	    private int getKhNumber(String command) {
   358	        int ch;
   359	        try {
   360	            ch = Integer.parseInt(command.substring(3));
   361	        }
   362	        catch (NumberFormatException ne) {
   363	            ch = -1;
   364	        }
   365	        if (ch <= 1) return ch;
   366	        if (acceptedPrimePower(ch)) return ch;
   367	        return -1;
   368	    }
   369	
   370	    private boolean acceptedPrimePower(int ch) {
   371	        boolean accepted = false;
   372	        int i = 0;
   373	        while (!accepted && i < options.getPrimes().size()) {
   374	            int prime = options.getPrimes().get(i);
   375	            if (ch%prime == 0 && powerOkay(ch, prime)) accepted = true;
   376	            else i++;
   377	        }
   378	        return accepted;
   379	    }
   380	
   381	    private boolean powerOkay(int ch, int prime) {
   382	        int maxPower = options.getPowers().get(options.getPrimes().indexOf(prime));
   383	        int i = 1;
   384	        int p = prime;
   385	        boolean okay = false;
   386	        while (!okay && i <= maxPower) {
   387	            if (ch == p) okay = true;
   388	            else {
   389	                i++;
   390	                p = p * prime;
   391	            }
   392	        }
   393	        return okay;
   394	    }

   396	    private void removeRedundancies(ArrayList<Integer> khovs) {
   397	        if (khovs.contains(0)) {
   398	            if (khovs.size() == 1) return;
   399	            khovs.clear();
   400	            khovs.add(0);
   401	            return;
   402	        }
   403	        for (int prime : options.getPrimes()) {
   404	            ArrayList<Integer> powers = new ArrayList<Integer>();
   405	            for (int u : khovs) {
   406	                if (u % prime == 0) powers.add(u);
   407	            }
   408	            Collections.sort(powers);
   409	            for (int i = 0; i < powers.size()-1; i++) khovs.remove(powers.get(i));
   410	        }
   411	    }

   413	    private void calculate() {
   414	        if (help) {
   415	            schreiNachHilfe();
   416	            return;
   417	        }
   418	        if (fileNames.isEmpty()) { // if there are no files, you can enter a pd-code, or hit return to end the program.
   419	            printFile = false;
   420	            try (BufferedReader reader = new BufferedReader(new InputStreamReader(System.in))) {
   421	                String linkString;
   422	                while (((linkString=reader.readLine())!=null) & (linkString.length() != 0)){
   423	                    if (linkString.charAt(0) == '-') {
   424	                        resetCommands();
   425	                        arguments = linkString.split("\\s+");
   426	                        getTheCommandsAndFiles();
   427	                        identifyCommands();
   428	                    }
   429	                    else {
   430	                        readOneLink(linkString);
   431	                        doCalculations(false);
   432	                        printTheStuff(0, null);
   433	                        theLinks.remove(0);
   434	                    }
   435	                }
   436	            }
   437	             catch (IOException ex) {
   438	                Logger.getLogger(KnotJobCommand.class.getName()).log(Level.SEVERE, null, ex);
   439	             }
   440	        }
   441	        else {
   442	            doCalculations(true);
   443	            printOutInfo();
   444	            if (!filters.isEmpty()) System.out.println("Total number : "+filteredCount);
   445	        }
   446	    }

   566	            for (int co : xtorts) {
   567	                if (extra) extraInfo = "X-Tortion mod "+co+" of file "
   568	                                +fileNames.get(count)+" finished.";
   569	                SpectralCalculator calculator = new SpectralCalculator(links, options, 
   570	                        new DialogWrap(countDown, extraInfo), co, true);
   571	                calculator.start();
   572	            }
   573	            for (int co : sltKhovs) {
   574	                if (extra) extraInfo = "sl_3 Kh mod "+co+" of file "
   575	                                +fileNames.get(count)+" finished.";
   576	                SlThreeHomCalculator calculator = new SlThreeHomCalculator(links, co, options, 
   577	                        new DialogWrap(countDown, extraInfo));
   578	                calculator.start();
   579	            }
   580	            for (int ch : slInvs) {
   581	                if (extra) extraInfo = "sl_3 S-invariant mod "+ch+" of file "
   582	                                +fileNames.get(count)+" finished.";
   583	                SlTSInvariantCalculator calcS = new SlTSInvariantCalculator(links, ch, false, options, 
   584	                        new DialogWrap(countDown, extraInfo));
   585	                calcS.start();
   586	            }
   587	        }
   588	        try {
   589	            countDown.await(); // all these calculators are threads. Here we wait 
   590	            // until they are all finished.
   591	        } 
   592	        catch (InterruptedException ex) {
   593	            Logger.getLogger(KnotJobCommand.class.getName()).log(Level.SEVERE, null, ex);
   594	        }
   595	    }

   621	    private void khovanovOutput(HomologyInfo info, String extra, boolean doubl, PrintWriter pw, int k) {
   622	        printOut(extra, info.poincarePolynomial().toString().substring(1), pw);
   623	        ArrayList<PoincarePolynomial> torPolys = info.torsionPolynomials(doubl);
   624	        if (csvFile) {
   625	            ArrayList<Integer> tors = new ArrayList<Integer>();
   626	            if (k == 0) tors = allTorsionU;
   627	            if (k == 1) tors = allTorsionR;
   628	            if (k == 2) tors = allTorsionO;
   629	            if (k == 3) tors = allTorsionSlTU;
   630	            if (k == 4) tors = allTorsionSlTR;
   631	            for (int t : tors) {
   632	                PoincarePolynomial poly = thePoly(t, torPolys);
   633	                if (poly != null) printOut("Torsion of order "+poly.torsion()+" : ", poly.toString().substring(1), pw);
   634	                else printOut(",", " ", pw);
   635	            }
   636	        } 
   637	        else {
   638	            for (PoincarePolynomial poly : torPolys) {
   639	                printOut("Torsion of order "+poly.torsion()+" :", poly.toString(), pw);
   640	            }
   641	        }
   642	    }

   644	    private PoincarePolynomial thePoly(int t, ArrayList<PoincarePolynomial> torPolys) {
   645	        int k = 0;
   646	        while (k < torPolys.size()) {
   647	            if (torPolys.get(k).torsion().intValue() == t) return torPolys.get(k);
   648	            k++;
   649	        }
   650	        return null;
   651	    }

   678	    private void printData(LinkData data, PrintWriter pw) {
   679	        filteredCount++;
   680	        printOut("", data.name, pw);
   681	        for (int s : sInvs) {
   682	            printOut("S-Invariant mod "+s+" : ", data.sInvariant(s).toString(), pw);
   683	        }
   684	        for (int s : slInvs) {
   685	            printOut("sl_3 S-Invariant mod "+s+" : ", data.sSlTInvariant(s, false).toString(), pw);
   686	        }
   687	        if (sqOneEven) printOut("Even Sq^1 Invariant : ", data.sqEven, pw);
   688	        if (sqOneOdd) printOut("Odd Sq^1 Invariant : ", data.sqOdd, pw);
   689	        if (sqOneSum) printOut("Sum Sq^1 Invariant : ", data.beta, pw);
   690	        if (comr) printOut("Complete reduced LS-Invariant : ", data.cmpinv, pw);
   691	        if (blsOdd) printOut("Odd BLS "+blsmod+" Invariant : ", data.bsOdd, pw);
   692	        if (sqTwoEven) printOut("Even Sq^2 Invariant : ", data.sqtEven, pw);
   693	        if (sqTwoOdd) printOut("Odd Sq^2 Invariant (eps=0) : ", data.sqtOdd, pw);
   694	        if (sqTwoOde) printOut("Odd Sq^2 Invariant (eps=1) : ", data.sqtOde, pw);
   695	        if (sGraded) printOut("Graded s-invariant : ", data.grsinv, pw);
   696	        if (signature) printOut("Signature : ", data.signature, pw);
   697	        for (int pp : evenKhovs) {
   698	            if (pp == 0) {
   699	                if (khevenunr) khovanovOutput(data.integralKhovHomology(false), 
   700	                        "Integral unreduced Khovanov Homology : ", false, pw, 0);
   701	                if (khevenred) khovanovOutput(data.integralKhovHomology(true), 
   702	                        "Integral reduced Khovanov Homology : ", false, pw, 1);
   703	            }
   704	            if (pp == 1) {
   705	                if (khevenunr) khovanovOutput(data.rationalKhovHomology(false), 
   706	                        "Rational unreduced Khovanov Homology : ", false, pw, -1);
   707	                if (khevenred) khovanovOutput(data.rationalKhovHomology(true), 
   708	                        "Rational reduced Khovanov Homology : ", false, pw, -1);
   709	            }
   710	            if (pp > 1) {
   711	                if (khevenunr) khovanovOutput(data.modKhovHomology(false, pp), 
   712	                        "Unreduced Khovanov Homology mod "+pp+" : ", true, pw, -1);
   713	                if (khevenred) khovanovOutput(data.modKhovHomology(true, pp), 
   714	                        "Reduced Khovanov Homology mod "+pp+" : ", true, pw, -1);
   715	            }
   716	        }
   717	        for (int pp : oddKhovs) {
   718	            if (pp == 0) {
   719	                khovanovOutput(data.integralOddKhHomology(), 
   720	                        "Odd integral Khovanov Homology : ", false, pw, 2);
   721	            }
   722	            if (pp == 1) {
   723	                khovanovOutput(data.rationalOddKhHomology(), 
   724	                        "Odd rational Khovanov Homology : ", false, pw, -1);
   725	            }
   726	            if (pp > 1) {
   727	                khovanovOutput(data.modOddKhHomology(pp), 
   728	                        "Odd Khovanov Homology mod "+pp+" : ", true, pw, -1);
   729	            }
   730	        }
   731	        for (int pp : sltKhovs) {
   732	            if (pp == 0) {
   733	                khovanovOutput(data.integralSlTHomology(false), 
   734	                        "Unreduced integral sl_3 Homology : ", false, pw, 3);
   735	                khovanovOutput(data.integralSlTHomology(true),
   736	                        "Reduced integral sl_3 Homology : ", false, pw, 4);
   737	            }
   738	            /*if (pp == 1) { // to follow
   739	                khovanovOutput(data.rationalOddKhHomology(), 
   740	                        "Odd rational Khovanov Homology : ", false, pw, -1);
   741	            }
   742	            if (pp > 1) {
   743	                khovanovOutput(data.modOddKhHomology(pp), 
   744	                        "Odd Khovanov Homology mod "+pp+" : ", true, pw, -1);
   745	            }// */
   746	        }
   747	        for (int co : xtorts) {
   748	            printOut("Extortion Order mod "+co+" : ", data.xtortion(co), pw);
   749	        }
   750	        if (csvFile) pw.println();
   751	        if (printCompDones) printOut("computation done", "", pw);
   752	    }

   795	    private void getTorsionFrom(int i, int kind) {
   796	        if (kind <= 1) {
   797	            boolean sign = kind == 1;
   798	            for (LinkData data : theLinks.get(i)) {
   799	                HomologyInfo info = data.integralKhovHomology(sign);
   800	                ArrayList<PoincarePolynomial> torPolys = info.torsionPolynomials(false);
   801	                for (PoincarePolynomial pol : torPolys) {
   802	                    int t = pol.torsion().intValue();
   803	                    if (kind == 0 && !allTorsionU.contains(t)) allTorsionU.add(t);
   804	                    if (kind == 1 && !allTorsionR.contains(t)) allTorsionR.add(t);
   805	                }
   806	            }
   807	        }
   808	        if (kind == 2) {
   809	            for (LinkData data : theLinks.get(i)) {
   810	                HomologyInfo info = data.integralOddKhHomology();
   811	                ArrayList<PoincarePolynomial> torPolys = info.torsionPolynomials(false);
   812	                for (PoincarePolynomial pol : torPolys) {
   813	                    int t = pol.torsion().intValue();
   814	                    if (!allTorsionO.contains(t)) allTorsionO.add(t);
   815	                }
   816	            }
   817	        }
   818	        if (kind == 3 || kind == 4) {
   819	            boolean red = false;
   820	            ArrayList<Integer> allTor = allTorsionSlTU;
   821	            if (kind == 4) {
   822	                red = true;
   823	                allTor = allTorsionSlTR;
   824	            }
   825	            for (LinkData data : theLinks.get(i)) {
   826	                HomologyInfo info = data.integralSlTHomology(red);
   827	                ArrayList<PoincarePolynomial> torPolys = info.torsionPolynomials(false);
   828	                for (PoincarePolynomial pol : torPolys) {
   829	                    int t = pol.torsion().intValue();
   830	                    if (!allTor.contains(t)) allTor.add(t);
   831	                }
   832	            }
   833	        }
   834	    }

   836	    private String newFileName(String name) {
   837	        for (int s : sInvs) name = name+"_s"+s;
   838	        for (int s : slInvs) name = name+"_sl3s"+s;
   839	        if (sqOneEven) name = name+"_sqe";
   840	        if (sqOneOdd) name = name+"_sqo";
   841	        if (signature) name = name+"_sgn";
   842	        String kh = "_kb";
   843	        if (!khevenred) kh = "_ku";
   844	        if (!khevenunr) kh = "_kr";
   845	        for (int p : evenKhovs) name = name+kh+p;
   846	        for (int p : oddKhovs) name = name+"_ko"+p;
   847	        if (csvFile) {
   848	            int u = name.indexOf(".");
   849	            name = name.substring(0, u)+".csv";
   850	            if (csvDirectory != null) {
   851	                int v = name.lastIndexOf(options.getSlash());
   852	                name = csvDirectory+name.substring(v+1);
   853	            }
   854	        }
   855	        return name;
   856	    }
